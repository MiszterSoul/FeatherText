Install
Package or static assets.
Run npm test and npm run build before publishing.

      
npm install feathertext

import FeatherText from 'feathertext';
import 'feathertext/css';

new FeatherText('#editor', {
  theme: 'dark',
  fancy: false
});
<link rel="stylesheet" href="feathertext.css">
<script src="feathertext.js"></script>
<textarea id="editor"></textarea>
<script>
  FeatherText.init('#editor', { fancy: false });
</script>
